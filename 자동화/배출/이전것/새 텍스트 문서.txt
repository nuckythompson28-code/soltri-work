import os
from datetime import datetime, timedelta
from docx import Document
import copy

def replace_text(doc, old_text, new_text):
    """문서 내의 모든 텍스트(표 포함)를 찾아 교체하는 함수"""
    for p in doc.paragraphs:
        if old_text in p.text:
            p.text = p.text.replace(old_text, new_text)
    
    for table in doc.tables:
        for row in table.rows:
            for cell in row.cells:
                for paragraph in cell.paragraphs:
                    if old_text in paragraph.text:
                        paragraph.text = paragraph.text.replace(old_text, new_text)

def generate_2026_records():
    # 1. 원본 템플릿 불러오기
    template_path = 'template.docx' 
    if not os.path.exists(template_path):
        print("에러: template.docx 파일이 같은 폴더에 있어야 합니다.")
        return

    # 2. 저장할 폴더 만들기
    output_dir = "2026_운영기록부_결과물"
    if not os.path.exists(output_dir):
        os.makedirs(output_dir)

    start_date = datetime(2026, 1, 1)
    end_date = datetime(2026, 12, 31)
    days_ko = ["월", "화", "수", "목", "금", "토", "일"]

    current_date = start_date
    while current_date <= end_date:
        # 날짜 데이터 준비
        yymmdd = current_date.strftime('%y%m%d')
        yyyy_val = str(current_date.year)
        mm_val = str(current_date.month)
        dd_val = str(current_date.day)
        wd_val = days_ko[current_date.weekday()]

        # 템플릿 다시 열기 (매번 새로 복사본 생성)
        doc = Document(template_path)

        # 문서 내용 교체 (날짜 및 요일)
        replace_text(doc, "YYYY", yyyy_val)
        replace_text(doc, "MM", mm_val)
        replace_text(doc, "DD", dd_val)
        replace_text(doc, "WD", wd_val)

        # 파일명 결정: 배출방지시설운영기록부(260101) - 목.docx
        save_name = f"배출방지시설운영기록부({yymmdd}) - {wd_val}.docx"
        doc.save(os.path.join(output_dir, save_name))
        
        print(f"{save_name} 생성 중...")
        current_date += timedelta(days=1)

    print("--- 모든 파일 생성 완료 ---")

if __name__ == "__main__":
    generate_2026_records()
